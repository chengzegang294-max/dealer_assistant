# 来源: 神奇数字系列指标[金鼎].doc
# ============================================================

CHI:=5;
KUAI:=13;
ZHONG:=21;
HUAN:=34;
MAN:=55;
DUN:=89;
JIE:=144;
快线:EMA(CLOSE,KUAI),COLORMAGENTA;
慢线:EMA(CLOSE,MAN),COLORYELLOW;
界线:EMA(CLOSE,JIE),COLORBLUE,LINETHICK2;
WR1:=((100*(HHV(HIGH,DUN)-CLOSE))/(HHV(HIGH,DUN)-LLV(LOW,DUN)));
WR2:=((100*(HHV(HIGH,HUAN)-CLOSE))/(HHV(HIGH,HUAN)-LLV(LOW,HUAN)));
E反向WR1:=EMA(100-WR1,CHI);
EE反向WR2:=EMA(EMA(100-WR2,JIE),CHI);
走强:=E反向WR1>EE反向WR2;
走弱:=E反向WR1<EE反向WR2;
STICKLINE(走弱,CLOSE,OPEN,3,1),COLORGREEN;
STICKLINE(走弱,LOW,(MIN(OPEN,CLOSE) * 1.00),0,0),COLORGREEN;
STICKLINE(走弱,MAX(OPEN,CLOSE),HIGH,0,0),COLORGREEN;
STICKLINE(走强,OPEN,CLOSE,3,1),COLORRED;
STICKLINE(走强,HIGH,MAX(CLOSE,OPEN),0,0),COLORRED;
STICKLINE(走强,LOW,(MIN(CLOSE,OPEN) * 1.00),0,0),COLORRED;
STICKLINE(((OPEN < CLOSE) AND 走强),(OPEN * 1),(CLOSE * 1.00),3,1),COLORRED;
STICKLINE(((OPEN > CLOSE) AND 走强),(OPEN * 1),(CLOSE * 1.00),3,0),COLORRED;
STICKLINE(((OPEN < CLOSE) AND 走弱),(CLOSE * 1),(OPEN * 1.00),3,1),COLORGREEN;
STICKLINE(((OPEN > CLOSE) AND 走弱),CLOSE,OPEN,3,0),COLORGREEN;
