BEN:=3;
CHI:=5;
KUAI:=13;
ZHONG:=21;
HUAN:=34;
MAN:=55;
DUN:=89;
JIE:=144;
MTR:=MAX1((HIGH-LOW),ABS(REF(CLOSE,1)-HIGH),ABS(REF(CLOSE,1)-LOW));
A3平均:=(EMA(MTR,HUAN)+EMA(MTR,MAN)+EMA(MTR,DUN))/3;
TYPICAL:=(2*CLOSE+HIGH+LOW)/4;
支撑线:(HHV(EMA(HIGH,KUAI),KUAI)+HHV(EMA(HIGH,ZHONG),ZHONG))/2-
3.142*A3平均,NODRAW;
压力线:(LLV(EMA(LOW,KUAI),KUAI)+LLV(EMA(LOW,ZHONG),ZHONG))/2+3.142*A3平均,NO
DRAW;
多空信号:=IF(CLOSE>REF(HHV(HIGH,89),1),1,IF(CLOSE<REF(LLV(LOW,89),1),-
1,0));
信号编译:=REF(多空信号,BARSLAST(多空信号));
多头状态:=信号编译=1;
空头状态:=信号编译=-1;
多头支撑:=EMA(TYPICAL,5)>支撑线 AND 多头状态;
空头压力:=EMA(TYPICAL,5)<压力线 AND 空头状态;
E4平均:=(EMA(CLOSE,ZHONG)+EMA(CLOSE,HUAN)+EMA(CLOSE,MAN)+EMA(CLOSE,DUN))/4;
S5:E4平均+5.500*A3平均,COLORBLUE,LINETHICK2;
卖货:E4平均+2.750*A3平均,RGB(128,64,0),LINETHICK2;
S1:E4平均+1.375*A3平均,COLORGRAY,DASH;
中轴:E4平均,COLORWHITE,LINETHICK2;
B1:E4平均-1.375*A3平均,COLORGRAY,DASH;
买货:E4平均-2.750*A3平均,RGB(128,64,0),LINETHICK2;
B5:E4平均-5.500*A3平均,COLORBLUE,LINETHICK2;
PARTLINE(多头支撑,支撑线,COLORMAGENTA),LINETHICK3;
PARTLINE(空头压力,压力线,COLORYELLOW),LINETHICK3;
