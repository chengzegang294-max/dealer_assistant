CHI:=5;
KUAI:=13;
HUAN:=34;
MAN:=55;
DUN:=89;
JIE:=144;
快线:EMA(CLOSE,KUAI),RGB(255,128,0),LINETHICK3;
慢线:EMA(CLOSE,MAN),RGB(128,0,255),LINETHICK3;
平衡点:(EMA(CLOSE,30)+EMA(CLOSE,72))/2,COLORBLACK,DOT;
沽货3:平衡点*1.05,RGB(128,0,0),LINETHICK2;
沽货2:平衡点*1.03,COLORMAGENTA,LINETHICK2;
沽货1:平衡点*1.01,COLORBLACK;
揸货1:平衡点*0.99,COLORBLACK;
揸货2:平衡点*0.97,COLORMAGENTA,LINETHICK2;
揸货3:平衡点*0.95,RGB(128,0,0),LINETHICK2;
WR1:=((100*(HHV(HIGH,DUN)-CLOSE))/(HHV(HIGH,DUN)-LLV(LOW,DUN)));
WR2:=((100*(HHV(HIGH,HUAN)-CLOSE))/(HHV(HIGH,HUAN)-LLV(LOW,HUAN)));
E反向WR1:=EMA(100-WR1,CHI);
EE反向WR2:=EMA(EMA(100-WR2,JIE),CHI);
走强:=E反向WR1>EE反向WR2;
走弱:=E反向WR1<EE反向WR2;
DRAWCOLORKLINE(走弱 AND CLOSE>揸货2 AND CLOSE>OPEN,RGB(0,120,0),1);
DRAWCOLORKLINE(走弱 AND CLOSE>揸货2 AND CLOSE<=OPEN,RGB(0,120,0),0);
DRAWCOLORKLINE(走强 AND CLOSE<沽货2 AND CLOSE>OPEN,COLORRED,1);
DRAWCOLORKLINE(走强 AND CLOSE<沽货2 AND CLOSE<=OPEN,COLORRED,0);
DRAWCOLORKLINE(走弱 AND CLOSE<=揸货2 AND CLOSE>OPEN,COLORBLUE,1);
DRAWCOLORKLINE(走弱 AND CLOSE<=揸货2 AND CLOSE<=OPEN,COLORBLUE,0);
DRAWCOLORKLINE(走强 AND CLOSE>=沽货2 AND CLOSE>OPEN,COLORBLUE,1);
DRAWCOLORKLINE(走强 AND CLOSE>=沽货2 AND CLOSE<=OPEN,COLORBLUE,0);
DRAWCOLORKLINE(走弱 AND CLOSE<=揸货3 AND CLOSE>OPEN,COLORGRAY,1);
DRAWCOLORKLINE(走弱 AND CLOSE<=揸货3 AND CLOSE<=OPEN,COLORGRAY,0);
DRAWCOLORKLINE(走强 AND CLOSE>=沽货3 AND CLOSE>OPEN,COLORGRAY,1);
DRAWCOLORKLINE(走强 AND CLOSE>=沽货3 AND CLOSE<=OPEN,COLORGRAY,0);
