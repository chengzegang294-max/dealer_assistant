# 明镜非台
# 来源: 神奇数字系列指标[金鼎].doc
# ============================================================

BEN:=3;
XUN:=8;
KUAI:=13;
ZHONG:=21;
MAN:=55;
JIE:=144;
快线:=EMA(CLOSE,KUAI);
中线:=EMA(CLOSE,ZHONG);
TYP:=(HIGH+LOW+CLOSE)/3;
CCI:=(TYP-EMA(TYP,JIE))/(0.015*AVEDEV(TYP,JIE));
DIFF:=快线-中线;
DEA:=EMA(DIFF,XUN);
M分水岭:=EMA(ABS(DIFF),JIE);
M揸货点:=DIFF<-1.96*M分水岭 AND CROSS(DIFF,DEA) AND CCI<-JIE;
M沽货点:=DIFF>1.96*M分水岭 AND CROSS(DEA,DIFF) AND CCI>JIE;
LC:=REF(CLOSE,1);
RSI1:=SMA(MAX(CLOSE-LC,0),XUN,1)/SMA(ABS(CLOSE-LC),XUN,1)*100;
RSI2:=SMA(MAX(CLOSE-LC,0),KUAI,1)/SMA(ABS(CLOSE-LC),KUAI,1)*100;
R最小值:=MIN(RSI1,RSI2);
R最大值:=MAX(RSI1,RSI2);
R揸货点:=R最大值<30 AND CROSS(RSI1,RSI2) AND CCI<-JIE;
R沽货点:=R最小值>70 AND CROSS(RSI2,RSI1) AND CCI>JIE;
RSV快:=(CLOSE-LLV(LOW,KUAI))/(HHV(HIGH,KUAI)-LLV(LOW,KUAI))*100;
K快:=SMA(RSV快,BEN,1);
RSV慢:=(CLOSE-LLV(LOW,MAN))/(HHV(HIGH,MAN)-LLV(LOW, MAN))*100;
K慢:=SMA(RSV慢,KUAI,1);
K最小值:=MIN(K快,K慢);
K最大值:=MAX(K快,K慢);
K揸货点:=K最大值<30 AND CROSS(K快,K慢) AND CCI<-JIE;
K沽货点:=K最小值>70 AND CROSS(K慢,K快) AND CCI>JIE;
MTR:=MAX(MAX((HIGH-LOW),ABS(REF(CLOSE,1)-HIGH)),ABS(REF(CLOSE,1)-LOW));
ATR:=EMA(MTR,KUAI);
DRAWTEXT(M揸货点,(LOW-0.382*ATR),'揸M'),COLORFFA500;
DRAWTEXT(M沽货点,(HIGH+0.618*ATR),'沽M'),COLORA020F0;
DRAWTEXT(R揸货点,(LOW-0.618*ATR),'揸R'),COLORGREEN;
DRAWTEXT(R沽货点,(HIGH+1.000*ATR),'沽R'),COLORGREEN;
DRAWTEXT(K揸货点,(LOW-1.000*ATR),'揸K'),COLORWHITE;
DRAWTEXT(K沽货点,(HIGH+1.382*ATR),'沽K'),COLORWHITE;
