BEN:=3;
XUN:=8;
KUAI:=13;
ZHONG:=21;
HUAN:=34;
MAN:=55;
DUN:=89;
JIE:=144;
TYPICAL:=(2*CLOSE+HIGH+LOW)/4;
CCI:=DUN*(TYPICAL-EMA(TYPICAL,MAN))/(AVEDEV(TYPICAL,MAN));
P最小值:=LLV(LOW,HUAN);
P最大值:=HHV(HIGH,HUAN);
散户:=EMA((TYPICAL-P最小值)/(P最大值-P最小值)*100,KUAI);
新庄:=EMA(0.618*REF(散户,1)+0.382*散户,2);
诞生:=CROSS(散户,新庄) AND 散户<30;
上扬:=CROSS(散户,30) AND 新庄<散户;
叹息:=CROSS(散户,65) AND 新庄<散户;
套牢:=CROSS(散户,75) AND 新庄<散户;
毁灭:=CROSS(新庄,散户) AND 新庄>75;
震荡状态:=套牢 OR 叹息 OR 上扬;
PBT:=BARSLAST(CROSS(COUNT(CROSS(CCI,-JIE),0)=1,0.5))+1;
PST:=BARSLAST(CROSS(COUNT(CROSS(JIE,CCI),0)=1,0.5))+1;
P买点:=诞生 AND BARSLAST(诞生)<PBT;
P卖点:=毁灭 AND BARSLAST(毁灭)<PST;
PBCOUNT:=COUNT(P买点,BARSLAST(震荡状态 OR 毁灭));
PSCOUNT:=COUNT(P卖点,BARSLAST(震荡状态 OR 诞生));
P揸货点:=P买点 AND PBCOUNT>=2 AND EVERY(CCI<-JIE,SUMBARS(PBCOUNT,2));
P沽货点:=P卖点 AND PSCOUNT>=2 AND EVERY(CCI>JIE,SUMBARS(PSCOUNT,2));
RSV快:=(CLOSE-LLV(LOW,KUAI))/(HHV(HIGH,KUAI)-LLV(LOW,KUAI))*100;
K快:=SMA(RSV快,BEN,1);
D快:=SMA(K快,BEN,1);
RSV慢:=(CLOSE-LLV(LOW,MAN))/(HHV(HIGH,MAN)-LLV(LOW,MAN))*100;
K慢:=SMA(RSV慢,KUAI,1);
D慢:=SMA(K慢,KUAI,1);
K最小值:=MIN1(K快,D快,K慢,D慢);
K最大值:=MAX1(K快,D快,K慢,D慢);
K揸货点:=K最大值<30 AND CROSS(K快,K慢) AND CCI<-JIE;
K沽货点:=K最小值>70 AND CROSS(K慢,K快) AND CCI>JIE;
快线:=EMA(CLOSE,KUAI);
中线:=EMA(CLOSE,ZHONG);
DIFF:=快线-中线;
DEA:=EMA(DIFF,XUN);
MACD2平均:=(DIFF+DEA)/2;
M分水岭:=EMA(ABS(MACD2平均),JIE);
M揸货点:=DIFF<-1.960*M分水岭 AND CROSS(DIFF,DEA) AND CCI<-JIE;
M沽货点:=DIFF>1.960*M分水岭 AND CROSS(DEA,DIFF) AND CCI>JIE;
RSI1:=SMA(MAX(CLOSE-REF(CLOSE,1),0),XUN,1)/SMA(ABS(CLOSE-
REF(CLOSE,1)),XUN,1)*100;
RSI2:=SMA(MAX(CLOSE-REF(CLOSE,1),0),KUAI,1)/SMA(ABS(CLOSE-
REF(CLOSE,1)),KUAI,1)*100;
RSI3:=SMA(MAX(CLOSE-REF(CLOSE,1),0),ZHONG,1)/SMA(ABS(CLOSE-
REF(CLOSE,1)),ZHONG,1)*100;
R最小值:=MIN1(RSI1,RSI2,RSI3);
R最大值:=MAX1(RSI1,RSI2,RSI3);
R揸货点:=BARSLASTCOUNT(R最大值<30  AND  RSI1>RSI2)>=1  AND  CROSS(RSI1,RSI3)
AND CCI<-JIE;
R沽货点:=BARSLASTCOUNT(R最小值>70  AND  RSI1<RSI2)>=1  AND  CROSS(RSI3,RSI1)
AND CCI>JIE;
MTR:=MAX(MAX((HIGH-LOW),ABS(REF(CLOSE,1)-HIGH)),ABS(REF(CLOSE,1)-LOW));
ATR3平均:(EMA(MTR,HUAN)+EMA(MTR,MAN)+EMA(MTR,DUN))/3,NODRAW;
揸P:=IF(P揸货点,1,0);
沽P:=IF(P沽货点,1,0);
揸K:=IF(K揸货点,1,0);
沽K:=IF(K沽货点,1,0);
揸M:=IF(M揸货点,1,0);
沽M:=IF(M沽货点,1,0);
揸R:=IF(R揸货点,1,0);
沽R:=IF(R沽货点,1,0);
揸沽:(揸P+揸K+揸M+揸R)-(沽P+沽K+沽M+沽R),NODRAW;
DRAWTEXT(P揸货点,(LOW-0.809*ATR3平均),'揸P'),COLORBLUE;
DRAWTEXT(P沽货点,(HIGH+0.809*ATR3平均),'沽P'),COLORBLUE;
DRAWTEXT(K揸货点,(LOW-0.618*ATR3平均),'揸K'),RGB(255,128,64);
DRAWTEXT(K沽货点,(HIGH+0.618*ATR3平均),'沽K'),RGB(128,128,255);
DRAWTEXT(M揸货点,(LOW-0.191*ATR3平均),'揸M'),COLORLIGHTRED;
DRAWTEXT(M沽货点,(HIGH+0.191*ATR3平均),'沽M'),RGB(0,120,0);
DRAWTEXT(R揸货点,(LOW-0.382*ATR3平均),'揸R'),COLORMAGENTA;
DRAWTEXT(R沽货点,(HIGH+0.382*ATR3平均),'沽R'),COLORYELLOW;
DRAWICON(P揸货点    OR    K揸货点    OR     M揸货点     OR     R揸货点,(LOW-
1.191*ATR3平均),'ICO1');
DRAWICON(P沽货点        OR        K沽货点        OR        M沽货点        OR
R沽货点,(HIGH+1.191*ATR3平均),'ICO2');
