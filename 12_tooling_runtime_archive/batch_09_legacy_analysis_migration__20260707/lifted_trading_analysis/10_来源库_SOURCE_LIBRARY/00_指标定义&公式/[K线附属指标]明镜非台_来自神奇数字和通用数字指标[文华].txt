JI:=2;
BEN:=3;
XUN:=8;
KUAI:=13;
ZHONG:=21;
HUAN:=34;
MAN:=55;
JIE:=144;
快线:=EMA(CLOSE,KUAI);
中线:=EMA(CLOSE,ZHONG);
TYP:=(HIGH+LOW+CLOSE)/3;
CCI:=(TYP-EMA(TYP,JIE))/(0.015*AVEDEV(TYP,JIE));
DIFF:=快线-中线;
DEA:=EMA(DIFF,XUN);
M分水岭:=EMA(ABS(DIFF),JIE);
M揸货点:=DIFF<-1.96*M分水岭 AND CROSS(DIFF,DEA) AND CCI<-JIE;
M沽货点:=DIFF>1.96*M分水岭 AND CROSS(DEA,DIFF) AND CCI>JIE;
LC:=REF(CLOSE,1);
RSI1:=SMA(MAX(CLOSE-LC,0),XUN,1)/SMA(ABS(CLOSE-LC),XUN,1)*100;
RSI2:=SMA(MAX(CLOSE-LC,0),KUAI,1)/SMA(ABS(CLOSE-LC),KUAI,1)*100;
RSI3:=SMA(MAX(CLOSE-LC,0),ZHONG,1)/SMA(ABS(CLOSE-LC),ZHONG,1)*100;
R最小值:=MIN1(RSI1,RSI2,RSI3);
R最大值:=MAX1(RSI1,RSI2,RSI3);
R揸货点:=BARSLASTCOUNT(R最大值<30  AND  RSI1>RSI2)>=1  AND  CROSS(RSI1,RSI3)
AND CCI<-JIE;
R沽货点:=BARSLASTCOUNT(R最小值>70  AND  RSI1<RSI2)>=1  AND  CROSS(RSI3,RSI1)
AND CCI>JIE;
RSV快:=(CLOSE-LLV(LOW,KUAI))/(HHV(HIGH,KUAI)-LLV(LOW,KUAI))*100;
K快:=SMA(RSV快,BEN,1),RGB(255,128,0);
D快:=SMA(K快,BEN,1),RGB(128,0,255);
RSV慢:=(CLOSE-LLV(LOW,MAN))/(HHV(HIGH,MAN)-LLV(LOW, MAN))*100;
K慢:=SMA(RSV慢,KUAI,1),COLORRED,LINETHICK2;
D慢:=SMA(K慢,KUAI,1),RGB(0,120,0),LINETHICK2;
K最小值:=MIN1(K快,D快,K慢,D慢);
K最大值:=MAX1(K快,D快,K慢,D慢);
K揸货点:=K最大值<30 AND CROSS(K快,K慢) AND CCI<-JIE;
K沽货点:=K最小值>70 AND CROSS(K慢,K快) AND CCI>JIE;
EMA加权:=(2*CLOSE+HIGH+LOW)/4;
E最低价:=LLV(LOW,HUAN);
E最高价:=HHV(HIGH,HUAN);
散户:=EMA((EMA加权-E最低价)/(E最高价-E最低价)*100,KUAI);
新庄:=EMA(0.618*REF(散户,1)+0.382*散户,JI);
诞生:=CROSS(散户,新庄) AND 散户<22;
上扬:=CROSS(散户,30) AND 新庄<散户;
叹息:=CROSS(散户,65) AND 新庄<散户;
套牢:=CROSS(散户,75) AND 新庄<散户;
毁灭:=CROSS(新庄,散户) AND 新庄>75;
震荡状态:=套牢 OR 叹息 OR 上扬;
EBT:=BARSLAST(CROSS(COUNT(CROSS(CCI,-JIE),0)=1,0.5))+1;
EST:=BARSLAST(CROSS(COUNT(CROSS(JIE,CCI),0)=1,0.5))+1;
E买点:=诞生 AND BARSLAST(诞生)<EBT;
E卖点:=毁灭 AND BARSLAST(毁灭)<EST;
EBCOUNT:=COUNT(E买点,BARSLAST(震荡状态 OR 毁灭));
ESCOUNT:=COUNT(E卖点,BARSLAST(震荡状态 OR 诞生));
E揸货点:=E买点 AND EBCOUNT>=2 AND EVERY(CCI<-JIE,SUMBARS(EBCOUNT,2));
E沽货点:=E卖点 AND ESCOUNT>=2 AND EVERY(CCI>JIE,SUMBARS(ESCOUNT,2));
M揸:=IF(M揸货点,1,0);
M沽:=IF(M沽货点,1,0);
R揸:=IF(R揸货点,1,0);
R沽:=IF(R沽货点,1,0);
K揸:=IF(K揸货点,1,0);
K沽:=IF(K沽货点,1,0);
E揸:=IF(E揸货点,1,0);
E沽:=IF(E沽货点,1,0);
MTR:=MAX(MAX((HIGH-LOW),ABS(REF(CLOSE,1)-HIGH)),ABS(REF(CLOSE,1)-LOW));
ATR:=EMA(MTR,KUAI);
DRAWTEXT(M揸货点,(LOW-0.191*ATR),'揸M'),RGB(128,0,0);
DRAWTEXT(M沽货点,(HIGH+0.191*ATR),'沽M'),COLORMAGENTA;
DRAWTEXT(R揸货点,(LOW-0.382*ATR),'揸R'),COLORRED;
DRAWTEXT(R沽货点,(HIGH+0.382*ATR),'沽R'),RGB(0,120,0);
DRAWTEXT(K揸货点,(LOW-0.618*ATR),'揸K'),RGB(255,128,0);
DRAWTEXT(K沽货点,(HIGH+0.618*ATR),'沽K'),RGB(128,0,255);
DRAWTEXT(E揸货点,(LOW-0.809*ATR),'揸E'),COLORBLUE;
DRAWTEXT(E沽货点,(HIGH+0.809*ATR),'沽E'),COLORBLUE;
DRAWICON(M揸货点 OR R揸货点 OR K揸货点 OR E揸货点,(LOW-1.191*ATR),'ICO1');
DRAWICON(M沽货点 OR R沽货点 OR K沽货点 OR E沽货点,(HIGH+1.191*ATR),'ICO2');
                               随机振荡器模板
AAAA:zz,NODRAW;
BBBB:yy,NODRAW;
CCCC:xx,NODRAW;
灵:=AAAA*10;
敏:=AAAA*3;
逐:=BBBB*10;
渐:=BBBB*3;
滞:=CCCC*10;
后:=CCCC*3;
RSV滞后:=(CLOSE-LLV(LOW,滞))/(HHV(HIGH,滞)-LLV(LOW,滞))*100;
K滞后:MA(RSV滞后,后),RGB(192,192,192),LINETHICK3;
D滞后:MA(K滞后,后),COLORBLACK,LINETHICK3;
RSV逐渐:=(CLOSE-LLV(LOW,逐))/(HHV(HIGH,逐)-LLV(LOW,逐))*100;
K逐渐:MA(RSV逐渐,渐),COLORRED,LINETHICK2;
D逐渐:MA(K逐渐,渐),RGB(0,120,0),LINETHICK2;
RSV灵敏:=(CLOSE-LLV(LOW,灵))/(HHV(HIGH,灵)-LLV(LOW,灵))*100;
K灵敏:MA(RSV灵敏,敏),RGB(255,128,0),DOT;
D灵敏:MA(K灵敏,敏),RGB(128,0,255),DOT;
KD滞后较小值:=MIN(K滞后,D滞后);
KD滞后较大值:=MAX(K滞后,D滞后);
KD逐渐较小值:=MIN(K逐渐,D逐渐);
KD逐渐较大值:=MAX(K逐渐,D逐渐);
低位金叉:=KD滞后较大值<30 AND KD逐渐较大值<20 AND CROSS(K逐渐,D逐渐);
高位死叉:=KD滞后较小值>70 AND KD逐渐较小值>80 AND CROSS(D逐渐,K逐渐);
KD更小值:=MIN(KD滞后较小值,KD逐渐较小值);
KD更大值:=MAX(KD滞后较大值,KD逐渐较大值);
DRAWICON(低位金叉,KD更小值,'ICO92');
DRAWICON(高位死叉,KD更大值,'ICO91');
REFLINE:20,50,80;
BACKGROUNDSTYLE(0);
